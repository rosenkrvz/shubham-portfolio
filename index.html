# Automated ML Classification Pipeline
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier

class ModelEngine:
    def __init__(self):
        self.clf = RandomForestClassifier(
            n_estimators=150,
            criterion='gini',
            random_state=42
        )

    def evaluate(self, X_train, y_train):
        self.clf.fit(X_train, y_train)
        accuracy = self.clf.score(X_train, y_train)
        return f"Accuracy: {accuracy:.2%}"
